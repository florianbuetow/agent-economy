_START = datetime(2026, 10, 17, 12, 0, tzinfo=UTC)
_BIDDING_SECONDS = 10
_ACCEPTANCE_WINDOW_SECONDS = 20
_WAIT_SECONDS = 10
_FUTURE_SECONDS = 15


class _GatewayEndpoint(Protocol):
    base_url: str


class _MutableDomainClock:
    """One synchronized task-domain clock for both transition actors."""

    def __init__(self, initial: datetime) -> None:
        self._value = initial
        self._lock = Lock()

    def now(self, timezone: tzinfo | None) -> datetime:
        with self._lock:
            value = self._value
        if timezone is None:
            return value.replace(tzinfo=None)
        return value.astimezone(timezone)

    def set(self, value: datetime) -> None:
        with self._lock:
            if value < self._value:
                raise AssertionError("MP-07 accept-expiry clock cannot move backwards")
            self._value = value


class _ClockDateTime:
    """datetime parsing with current time supplied by the domain clock."""

    clock: _MutableDomainClock

    @classmethod
    def now(cls, timezone: tzinfo | None = None) -> datetime:
        return cls.clock.now(timezone)

    @classmethod
    def fromisoformat(cls, value: str) -> datetime:
        return datetime.fromisoformat(value)


class _PostEvaluationGate:
    """Pause accept after its real pre-cutoff deadline evaluation."""

    def __init__(self, *, task_id: str) -> None:
        self._task_id = task_id
        self._ready = Event()
        self._release = Event()
        self._armed = True

    def pause(self, task: dict[str, Any]) -> None:
        if not self._armed or task.get("task_id") != self._task_id:
            return
        self._armed = False
        if task.get("status") != "open":
            raise AssertionError(
                f"MP-07 harness expected accept validation to retain open, got {task!r}"
            )
        self._ready.set()
        if not self._release.wait(timeout=_WAIT_SECONDS):
            raise AssertionError("MP-07 accept-expiry RELEASE gate timed out")

    def wait_until_ready(self) -> None:
        if not self._ready.wait(timeout=_WAIT_SECONDS):
            raise AssertionError("MP-07 accept-expiry READY gate timed out")

    def release(self) -> None:
        self._release.set()


class _GatedDeadlineEvaluator(DeadlineEvaluator):
    """Run real deadline logic before pausing the stale accept contender."""

    def __init__(
        self,
        *,
        store: TaskDbClient,
        escrow_coordinator: EscrowCoordinator,
        platform_agent: MagicMock,
        gate: _PostEvaluationGate,
    ) -> None:
        super().__init__(store, escrow_coordinator, platform_agent)
        self._gate = gate

    async def evaluate_deadline(self, task: dict[str, Any]) -> dict[str, Any]:
        evaluated = await super().evaluate_deadline(task)
        self._gate.pause(evaluated)
        return evaluated

def _install_domain_clock(monkeypatch: pytest.MonkeyPatch) -> _MutableDomainClock:
    clock = _MutableDomainClock(_START)
    _ClockDateTime.clock = clock
    for module_name in (
        "task_board_service.services.asset_manager",
        "task_board_service.services.deadline_evaluator",
        "task_board_service.services.task_bidding",
        "task_board_service.services.task_db_client",
        "task_board_service.services.task_helpers",
    ):
        monkeypatch.setattr(f"{module_name}.datetime", _ClockDateTime)
    return clock


def _read_rows(
    db_path: str,
    query: str,
    params: tuple[object, ...] = (),
) -> list[dict[str, object]]:
    connection = sqlite3.connect(db_path)
    connection.row_factory = sqlite3.Row
    try:
        return [dict(row) for row in connection.execute(query, params).fetchall()]
    finally:
        connection.close()


async def _seed_funded_escrow(
    gateway_client: httpx.AsyncClient,
    *,
    payer_id: str,
    task_id: str,
    escrow_id: str,
) -> dict[str, object]:
    await register_identity_agent(gateway_client, payer_id)
    account_response = await gateway_client.post(
        "/bank/accounts",
        json={
            "operation_id": f"op-{uuid.uuid4()}",
            "account_id": payer_id,
            "created_at": "2026-10-17T11:59:58Z",
            "balance": 100,
            "event": {
                "event_source": "central-bank",
                "event_type": "account.created",
                "timestamp": "2026-10-17T11:59:58Z",
                "agent_id": payer_id,
                "summary": f"Created account for {payer_id}",
                "payload": "{}",
            },
            "initial_credit": {
                "tx_id": f"tx-{uuid.uuid4()}",
                "amount": 100,
                "reference": "initial_balance",
                "timestamp": "2026-10-17T11:59:58Z",
            },
        },
    )
    assert account_response.status_code == 201, account_response.text

    lock_response = await gateway_client.post(
        "/bank/escrow/lock",
        json={
            "operation_id": operation_id_for("task-lock", (task_id, payer_id)),
            "escrow_id": escrow_id,
            "payer_account_id": payer_id,
            "amount": 100,
            "task_id": task_id,
            "created_at": "2026-10-17T11:59:59Z",
            "tx_id": f"tx-{uuid.uuid4()}",
            "event": {
                "event_source": "central-bank",
                "event_type": "escrow.locked",
                "timestamp": "2026-10-17T11:59:59Z",
                "agent_id": payer_id,
                "summary": f"Escrow locked for {task_id}",
                "payload": "{}",
            },
        },
    )
    assert lock_response.status_code == 201, lock_response.text
    envelope = lock_response.json()
    assert isinstance(envelope, dict)
    assert envelope["outcome"] == "native_receipt"
    return envelope


async def _create_open_task_with_bid(
    client: httpx.AsyncClient,
    gateway_client: httpx.AsyncClient,
    poster_key: Ed25519PrivateKey,
    worker_key: Ed25519PrivateKey,
) -> _PublicSetup:
    poster_id = f"a-mp07-accept-expiry-poster-{uuid.uuid4()}"
    worker_id = f"a-mp07-accept-expiry-worker-{uuid.uuid4()}"
    task_id = f"t-{uuid.uuid4()}"
    escrow_id = f"esc-{uuid.uuid4()}"
    await register_identity_agent(gateway_client, worker_id)
    lock_envelope = await _seed_funded_escrow(
        gateway_client,
        payer_id=poster_id,
        task_id=task_id,
        escrow_id=escrow_id,
    )
    state = get_app_state()
    assert state.central_bank_client is not None
    state.central_bank_client.escrow_lock = AsyncMock(return_value=lock_envelope)

    create_response = await client.post(
        "/tasks",
        json={
            "task_token": make_jws_token(
                poster_key,
                poster_id,
                {
                    "action": "create_task",
                    "task_id": task_id,
                    "poster_id": poster_id,
                    "title": "MP-07 acceptance versus expiry",
                    "spec": "Expire at the selected acceptance cutoff without stale acceptance.",
                    "reward": 100,
                    "bidding_deadline_seconds": _BIDDING_SECONDS,
                    "acceptance_window_seconds": _ACCEPTANCE_WINDOW_SECONDS,
                    "execution_deadline_seconds": 60,
                    "review_deadline_seconds": 60,
                },
            ),
            "escrow_token": make_jws_token(
                poster_key,
                poster_id,
                {
                    "action": "escrow_lock",
                    "operation_id": operation_id_for("task-lock", (task_id, poster_id)),
                    "task_id": task_id,
                    "agent_id": poster_id,
                    "amount": 100,
                },
            ),
        },
    )
    assert create_response.status_code == 201, create_response.text

    bid_response = await client.post(
        f"/tasks/{task_id}/bids",
        json={
            "token": make_jws_token(
                worker_key,
                worker_id,
                {
                    "action": "submit_bid",
                    "task_id": task_id,
                    "bidder_id": worker_id,
                    "amount": 90,
                },
            )
        },
    )
    assert bid_response.status_code == 201, bid_response.text
    return _PublicSetup(
        task_id=task_id,
        poster_id=poster_id,
        worker_id=worker_id,
        bid_id=str(bid_response.json()["bid_id"]),
        escrow_id=escrow_id,
    )


def _make_runtime(
    *,
    gateway: _GatewayEndpoint,
    gateway_db_path: str,
    asset_path: Path,
    cleanup: ExitStack,
    gate: _PostEvaluationGate | None,
) -> _BoardRuntime:
    key_dir = Path(gateway_db_path).parent
    signer = GatewayRequestSigner(
        GatewaySignerConfig(
            kid="gateway-task-board",
            private_key_path=key_dir / "gateway-task-board.key",
            audience="db-gateway",
            token_ttl_seconds=300,
        ),
        config_dir=key_dir,
    )
    store = TaskDbClient(base_url=gateway.base_url, timeout_seconds=10, signer=signer)
    cleanup.callback(store.close)

    platform = MagicMock()
    platform.agent_id = "a-platform-test-id"
    platform.validate_certificate = MagicMock(side_effect=verify_compact_jws)
    platform.file_claim = AsyncMock(
        return_value={"dispute_id": f"disp-{uuid.uuid4()}", "status": "rebuttal_pending"}
    )
    identity = AsyncMock()
    identity.verify_jws = AsyncMock(side_effect=verify_compact_jws)
    token_validator = TokenValidator(platform_agent=platform, identity_client=identity)

    bank = AsyncMock()
    bank.escrow_release = AsyncMock(return_value={"status": "recorded-boundary-only"})
    bank.escrow_split = AsyncMock(return_value={"status": "recorded-boundary-only"})
    escrow = EscrowCoordinator(central_bank_client=bank, store=store)
    if gate is None:
        deadline: DeadlineEvaluator = DeadlineEvaluator(store, escrow, platform)
    else:
        deadline = _GatedDeadlineEvaluator(
            store=store,
            escrow_coordinator=escrow,
            platform_agent=platform,
            gate=gate,
        )
    assets = AssetManager(
        store=store,
        token_validator=token_validator,
        deadline_evaluator=deadline,
        asset_storage_path=str(asset_path),
        max_file_size=10_485_760,
        max_files_per_task=20,
    )
    manager = TaskManager(
        store=store,
        central_bank_client=bank,
        escrow_coordinator=escrow,
        token_validator=token_validator,
        deadline_evaluator=deadline,
        asset_manager=assets,
        platform_signer=platform,
        platform_agent_id="a-platform-test-id",
    )
    return _BoardRuntime(manager, _RecordedBoundaries(bank, platform))


def _invoke_accept(
    call: Callable[[], Coroutine[Any, Any, dict[str, Any]]],
) -> _Outcome:
    try:
        return _Outcome(result=asyncio.run(call()), error=None)
    except ServiceError as exc:
        return _Outcome(result=None, error=exc)


def _evaluate_deadline(runtime: _BoardRuntime, task_id: str) -> dict[str, Any]:
    task = runtime.manager.store.get_task(task_id)
    if task is None:
        raise AssertionError(f"MP-07 harness could not reload task {task_id}")
    return asyncio.run(runtime.manager.deadline_evaluator.evaluate_deadline(task))


async def test_stale_validated_accept_cannot_overwrite_acceptance_expiry(
    client: httpx.AsyncClient,
    gateway_client: httpx.AsyncClient,
    poster_keypair: tuple[Ed25519PrivateKey, str],
    worker_keypair: tuple[Ed25519PrivateKey, str],
    gateway_db_path: str,
    _gateway: _GatewayEndpoint,
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    clock = _install_domain_clock(monkeypatch)
    setup = await _create_open_task_with_bid(
        client,
        gateway_client,
        poster_keypair[0],
        worker_keypair[0],
    )
    clock.set(_START + timedelta(seconds=_BIDDING_SECONDS, microseconds=-1))
    gate = _PostEvaluationGate(task_id=setup.task_id)

    with ExitStack() as cleanup:
        cleanup.callback(gate.release)
        contender = _make_runtime(
            gateway=_gateway,
            gateway_db_path=gateway_db_path,
            asset_path=tmp_path / "accept-contender-assets",
            cleanup=cleanup,
            gate=gate,
        )
        expiry = _make_runtime(
            gateway=_gateway,
            gateway_db_path=gateway_db_path,
            asset_path=tmp_path / "acceptance-expiry-assets",
            cleanup=cleanup,
            gate=None,
        )
        accept_token = make_jws_token(
            poster_keypair[0],
            setup.poster_id,
            {
                "action": "accept_bid",
                "operation_id": f"op-{uuid.uuid4()}",
                "expected_lifecycle_version": 0,
                "task_id": setup.task_id,
                "bid_id": setup.bid_id,
                "poster_id": setup.poster_id,
            },
        )
        executor = cleanup.enter_context(ThreadPoolExecutor(max_workers=2))
        accept_future = executor.submit(
            _invoke_accept,
            lambda: contender.manager.accept_bid(setup.task_id, setup.bid_id, accept_token),
        )
        try:
            await asyncio.to_thread(gate.wait_until_ready)
            acceptance_cutoff = _START + timedelta(
                seconds=_BIDDING_SECONDS + _ACCEPTANCE_WINDOW_SECONDS
            )
            clock.set(acceptance_cutoff)
            expiry_future = executor.submit(_evaluate_deadline, expiry, setup.task_id)
            expiry_result = expiry_future.result(timeout=_FUTURE_SECONDS)
            if expiry_result.get("status") != "expired":
                raise AssertionError(
                    f"MP-07 harness acceptance-expiry actor did not commit: {expiry_result!r}"
                )
        finally:
            gate.release()

        try:
            outcome = accept_future.result(timeout=_FUTURE_SECONDS)
        except Exception as exc:
            raise AssertionError(
                "MP-07 accept contender failed or timed out outside the proof"
            ) from exc

        task_rows = _read_rows(
            gateway_db_path,
            "SELECT status, poster_id, reward, escrow_id, worker_id, accepted_bid_id, "
            "accepted_at, expired_at FROM board_tasks WHERE task_id = ?",
            (setup.task_id,),
        )
        bid_rows = _read_rows(
            gateway_db_path,
            "SELECT bid_id, task_id, bidder_id, amount FROM board_bids WHERE task_id = ?",
            (setup.task_id,),
        )
        events = [
            str(row["event_type"])
            for row in _read_rows(
                gateway_db_path,
                "SELECT event_type FROM events WHERE task_id = ? "
                "AND event_type IN ('task.accepted', 'task.expired') ORDER BY event_id",
                (setup.task_id,),
            )
        ]
        diagnostics = {
            "outcome": {
                "result_status": outcome.result.get("status") if outcome.result else None,
                "error": outcome.error.error if outcome.error else None,
                "status_code": outcome.error.status_code if outcome.error else None,
            },
            "task_rows": task_rows,
            "bid_rows": bid_rows,
            "semantic_events": events,
            "expiry_bank_calls": expiry.boundaries.bank.escrow_release.await_count,
            "accept_bank_calls": contender.boundaries.bank.escrow_release.await_count,
            "accept_court_calls": contender.boundaries.court.file_claim.await_count,
        }

        if outcome.error is not None and outcome.error.status_code != 409:
            raise AssertionError(
                {"message": "Unexpected non-409 accept error; not the intended red", **diagnostics}
            )
        observed_task = task_rows[0] if len(task_rows) == 1 else None
        conflict_message = (
            "Stale accept must return 409 after acceptance expiry; "
            f"result_status={outcome.result.get('status') if outcome.result else None}, "
            f"error_status={outcome.error.status_code if outcome.error else None}, "
            f"final_status={observed_task.get('status') if observed_task else None}, "
            f"worker_id={observed_task.get('worker_id') if observed_task else None}, "
            f"accepted_bid_id={observed_task.get('accepted_bid_id') if observed_task else None}, "
            f"events={events}, expiry_bank_calls="
            f"{expiry.boundaries.bank.escrow_release.await_count}, "
            f"losing_bank_calls={contender.boundaries.bank.escrow_release.await_count}, "
            f"losing_court_calls={contender.boundaries.court.file_claim.await_count}"
        )
        assert outcome.error is not None and outcome.error.status_code == 409, conflict_message
        assert outcome.error.error == "task_version_conflict", diagnostics
        assert outcome.error.details["expected_lifecycle_version"] == 0, diagnostics
        assert outcome.error.details["actual_lifecycle_version"] == 1, diagnostics
        assert outcome.error.details["current_status"] == "expired", diagnostics
        assert len(task_rows) == 1, diagnostics
        task_row = task_rows[0]
        assert task_row["status"] == "expired", diagnostics
        assert task_row["poster_id"] == setup.poster_id, diagnostics
        assert task_row["reward"] == 100, diagnostics
        assert task_row["escrow_id"] == setup.escrow_id, diagnostics
        assert task_row["worker_id"] is None, diagnostics
        assert task_row["accepted_bid_id"] is None, diagnostics
        assert task_row["accepted_at"] is None, diagnostics
        expired_at = task_row["expired_at"]
        assert isinstance(expired_at, str), diagnostics
        assert datetime.fromisoformat(expired_at.replace("Z", "+00:00")) == acceptance_cutoff
        assert bid_rows == [
            {
                "bid_id": setup.bid_id,
                "task_id": setup.task_id,
                "bidder_id": setup.worker_id,
                "amount": 90,
            }
        ], diagnostics
        assert events == ["task.expired"], diagnostics
        assert expiry.boundaries.bank.escrow_release.await_count == 1, diagnostics
        assert contender.boundaries.bank.escrow_release.await_count == 0, diagnostics
        assert contender.boundaries.court.file_claim.await_count == 0, diagnostics
